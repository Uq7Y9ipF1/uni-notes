cmake_minimum_required(VERSION 3.10)
project(StaticExample LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(print_static STATIC util/print.cpp)

target_include_directories(print_static PUBLIC util)

add_executable(MyStaticProgram main.cpp)

target_link_libraries(MyStaticProgram PRIVATE print_static)
